function sendToWhatsApp() { const modeSelect = document.getElementById('modeView'); const modeText = modeSelect.options[modeSelect.selectedIndex].text; const type = modeSelect.value.includes('rct') ? 'RCT' : 'CMT'; // Parameter Header const grade = document.getElementById('grade').value; const jr = document.getElementById('jr').value || '-'; const temp = document.getElementById('temp').value || '-'; const gsm = document.getElementById('gsm').value; const rh = document.getElementById('rh').value || '-'; const tool = document.getElementById('tool').value; const faktor = document.getElementById('faktor').value; const minVal = document.getElementById('minValue').value; // Nilai Data Tabel const isReport = modeSelect.value.startsWith('report_'); const ts1 = document.getElementById('ts1').value || '-'; const ms1 = document.getElementById('ms1').value || '-'; const ms2 = document.getElementById('ms2').value || '-'; const ds1 = document.getElementById('ds1').value || '-'; const ts3 = document.getElementById('ts3').value || '-'; const ms3 = document.getElementById('ms3').value || '-'; const ds2 = document.getElementById('ds2').value || '-'; const ts4 = document.getElementById('ts4').value || '-'; const ms4 = document.getElementById('ms4').value || '-'; const ds3 = document.getElementById('ds3').value || '-'; // Hasil Perhitungan const avgTS = document.getElementById('avgTS').innerText; const avgMS = document.getElementById('avgMS').innerText; const avgDS = document.getElementById('avgDS').innerText; const grandAVG = document.getElementById('grandAVG').innerText; // Format Pesan Tabel untuk WhatsApp const message = `=========================\n` + ` *${modeText.toUpperCase()}*\n` + `=========================\n` + `*GRADE :* ${grade}\n` + `*JR :* ${jr} | *TEMP :* ${temp}°C\n` + `*GSM :* ${gsm} | *RH :* ${rh}%\n` + `*TOOL :* ${tool} | *F2 :* ${faktor}% | *MIN :* ${minVal}\n` + `-----------------------------------------\n` + `| *${type.padEnd(5)}* | *TS* | *MS* | *DS* |\n` + `-----------------------------------------\n` + `| TOP 1 | ${ts1.padEnd(6)} | ${ms1.padEnd(6)} | ${ds1.padEnd(6)} |\n` + `| TOP 2 | " | ${ms2.padEnd(6)} | " |\n` + `| BOT 3 | ${ts3.padEnd(6)} | ${ms3.padEnd(6)} | ${ds2.padEnd(6)} |\n` + `| BOT 4 | ${ts4.padEnd(6)} | ${ms4.padEnd(6)} | ${ds3.padEnd(6)} |\n` + `-----------------------------------------\n` + `| *AVE* | *${avgTS.padEnd(5)}* | *${avgMS.padEnd(5)}* | *${avgDS.padEnd(5)}* |\n` + `| *GRAND AVE :* *${grandAVG}*\n` + `-----------------------------------------\n` + `| *STATUS:* | Q1 | Q1 | Q1 |\n` + `=========================`; window.open(`https://wa.me/?text=${encodeURIComponent(message)}`, '_blank'); }
